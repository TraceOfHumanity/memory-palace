function plot(f, a, b, N, W, H, yMin, yMax) {
  const pts = [];
  for (let i = 0; i <= N; i++) {
    const x = a + (b - a) * i / N, y = f(x);
    if (!isFinite(y) || y < yMin || y > yMax) { pts.push(null); continue; }  // розрив
    pts.push([ (x - a) / (b - a) * W, H - (y - yMin) / (yMax - yMin) * H ]); // y вниз
  }
  return pts; // ламану рвати на null
}
