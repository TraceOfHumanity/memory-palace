# Euclid's Algorithm — алгоритм Евкліда та НСД

**Найбільший спільний дільник (НСД, GCD)** двох цілих — найбільше число, на яке діляться обидва. Алгоритм Евкліда (~300 р. до н. е.) — один із найстаріших **нетривіальних алгоритмів**, що досі використовується: у скороченні дробів, модульній арифметиці, криптографії (RSA), розв'язанні діофантових рівнянь. Його розширена версія знаходить також **коефіцієнти Безу** й **обернений елемент за модулем**.

## 1. Ідея

**Ключова властивість:** `gcd(a, b) = gcd(b, a mod b)`. Чому? Нехай `d` ділить `a` і `b`; тоді `d` ділить `a − k·b` для будь-якого `k`, зокрема `a mod b`. І навпаки: дільник `b` та `a mod b` ділить `a = k·b + (a mod b)`. Отже, множини спільних дільників пар `(a, b)` та `(b, a mod b)` збігаються — і НСД теж.

Крок зменшує числа; закінчується, коли друге число стає `0`: `gcd(a, 0) = a`.

Приклад: `gcd(48, 18)`: `48 mod 18 = 12` → `gcd(18, 12)` → `18 mod 12 = 6` → `gcd(12, 6)` → `12 mod 6 = 0` → `gcd(6, 0) = 6`. Кроки: `['gcd(48, 18) → 48 mod 18 = 12', 'gcd(18, 12) → 18 mod 12 = 6', 'gcd(12, 6) → 12 mod 6 = 0', 'gcd = 6']`; `gcd(48, 18)` = `6`.

## 2. Реалізація

```js
function gcd(a, b) {
  a = Math.abs(a);
  b = Math.abs(b);
  while (b !== 0) [a, b] = [b, a % b];
  return a;
}

function gcdRecursive(a, b) {
  return b === 0 ? Math.abs(a) : gcdRecursive(b, a % b);
}

function gcdTrace(a, b) {
  const out = [];
  while (b !== 0) {
    out.push('gcd(' + a + ', ' + b + ') → ' + a + ' mod ' + b + ' = ' + (a % b));
    [a, b] = [b, a % b];
  }
  out.push('gcd = ' + a);
  return out;
}

function gcdSteps(a, b) {
  let steps = 0;
  while (b !== 0) {
    [a, b] = [b, a % b];
    steps++;
  }
  return steps;
}

function lcm(a, b) {
  return a === 0 || b === 0 ? 0 : Math.abs(a / gcd(a, b) * b);
}

function gcdArray(arr) {
  return arr.reduce((g, x) => gcd(g, x), 0);
}

function lcmArray(arr) {
  return arr.reduce((l, x) => lcm(l, x), 1);
}

function extendedGcd(a, b) {
  if (b === 0) return { g: a, x: 1, y: 0 };
  const r = extendedGcd(b, a % b);
  return { g: r.g, x: r.y, y: r.x - Math.floor(a / b) * r.y };
}

function modInverse(a, m) {
  const r = extendedGcd(((a % m) + m) % m, m);
  if (r.g !== 1) return null;
  return ((r.x % m) + m) % m;
}

function solveDiophantine(a, b, c) {
  const r = extendedGcd(a, b);
  if (c % r.g !== 0) return null;
  const k = c / r.g;
  return { x: r.x * k, y: r.y * k, stepX: b / r.g, stepY: -a / r.g };
}

function binaryGcd(a, b) {
  if (a === 0) return b;
  if (b === 0) return a;
  let shift = 0;
  while (((a | b) & 1) === 0) { a >>= 1; b >>= 1; shift++; }
  while ((a & 1) === 0) a >>= 1;
  while (b !== 0) {
    while ((b & 1) === 0) b >>= 1;
    if (a > b) [a, b] = [b, a];
    b -= a;
  }
  return a << shift;
}

function reduceFraction(num, den) {
  const g = gcd(num, den);
  const sign = den < 0 ? -1 : 1;
  return [(sign * num) / g, (sign * den) / g];
}

function areCoprime(a, b) {
  return gcd(a, b) === 1;
}

function coprimeCountUpTo(n) {
  let c = 0;
  for (let i = 1; i <= n; i++) if (gcd(i, n) === 1) c++;
  return c;
}

function crt(remainders, moduli) {
  let x = 0;
  let M = 1;
  for (let i = 0; i < moduli.length; i++) {
    const m = moduli[i];
    const r = remainders[i];
    const inv = modInverse(M % m, m);
    if (inv === null) return null;
    const t = (((r - x) % m + m) % m * inv) % m;
    x += M * t;
    M *= m;
  }
  return { x: ((x % M) + M) % M, M };
}

function gcdBig(a, b) {
  a = a < 0n ? -a : a;
  b = b < 0n ? -b : b;
  while (b !== 0n) [a, b] = [b, a % b];
  return a;
}

function fibonacciWorstCase(n) {
  let a = 0;
  let b = 1;
  for (let i = 0; i < n; i++) [a, b] = [b, a + b];
  return [b, a];
}

function ratioOfRange(arr) {
  const g = gcdArray(arr);
  return arr.map((x) => x / g);
}

function gcdSubarraySum(arr) {
  return gcdArray(arr);
}

function pythagoreanTriples(limit) {
  const res = [];
  for (let m = 2; m * m < limit; m++) {
    for (let n = 1; n < m; n++) {
      if ((m - n) % 2 === 1 && gcd(m, n) === 1) {
        const a = m * m - n * n;
        const b = 2 * m * n;
        const c = m * m + n * n;
        if (c <= limit) res.push([Math.min(a, b), Math.max(a, b), c]);
      }
    }
  }
  return res.sort((x, y) => x[2] - y[2]);
}
```

## 3. Складність: теорема Ламе

Кількість кроків залежить від чисел. **Найгірший випадок** — послідовні числа Фібоначчі: `gcd(F_{n+1}, F_n)` робить `n − 1` кроків, бо кожен залишок — наступне менше число Фібоначчі. Теорема Ламе: кількість кроків для `a > b` не перевищує `5 × (кількість десяткових цифр `b`)`, тобто `O(log min(a, b))`.

Приклад: `fibonacciWorstCase(30)` — пара `[1346269, 832040]` дає 29 кроків; для випадкової пари великих чисел `gcdSteps(1000000007, 998244353)` = `9`. Для 64-бітових чисел — не більше ~92 кроків.

Операція `mod` на великих числах дорожча; для багаторозрядних чисел використовують `BigInt` (`gcdBig(2n ** 100n, 6n ** 50n)` = `1125899906842624n`).

## 4. НСК й властивості

- `lcm(a, b) = a / gcd(a, b) · b` (ділимо спершу, щоб не переповнити): `lcm(4, 6)` = `12`, `lcm(21, 6)` = `42`.
- `gcd(a, b) · lcm(a, b) = |a·b|`.
- `gcd(a, b, c) = gcd(gcd(a, b), c)`: `gcdArray([12, 18, 24])` = `6`, `lcmArray([2, 3, 4, 5])` = `60`.
- `gcd(0, 0) = 0` за домовленістю; `gcd(a, 0) = |a|`.
- **Взаємно прості** числа: `gcd = 1`: `areCoprime(14, 15)` = `true`.
- `gcd(a, b) = gcd(a, b − a)` (віднімання) — основа бінарного НСД (Штейна).

## 5. Розширений алгоритм Евкліда

Знаходить не лише `g = gcd(a, b)`, а й цілі `x, y` такі, що **`a·x + b·y = g`** (тотожність Безу). Рекурсія: якщо `b·x' + (a mod b)·y' = g`, то `a·y' + b·(x' − ⌊a/b⌋·y') = g`.

`extendedGcd(30, 18)` = `{ g: 6, x: -1, y: 2 }`: `30·(−1) + 18·2 = 6`.

### 5.1. Обернений елемент за модулем

Якщо `gcd(a, m) = 1`, то `a·x + m·y = 1` ⇒ `a·x ≡ 1 (mod m)`, тобто `x` — обернений до `a` за модулем `m`: `modInverse(3, 11)` = `4` (`3·4 = 12 ≡ 1`), `modInverse(10, 17)` = `12`, а `modInverse(4, 8)` = `null` (не існує). Основа **RSA** (`d = e⁻¹ mod φ(n)`) та ділення за простим модулем (альтернатива — мала теорема Ферма, [fast-exponentiation.md](fast-exponentiation.md)).

### 5.2. Лінійні діофантові рівняння

`a·x + b·y = c` має цілі розв'язки тоді й лише тоді, коли `gcd(a, b) | c`. Один розв'язок — `x₀ = x·(c/g)`, `y₀ = y·(c/g)`; усі — `(x₀ + k·b/g, y₀ − k·a/g)`. `solveDiophantine(6, 15, 9)` = `{ x: -6, y: 3, stepX: 5, stepY: -2 }`; `solveDiophantine(4, 6, 5)` = `null` (бо `gcd = 2 ∤ 5`).

### 5.3. Китайська теорема про залишки

Для взаємно простих модулів система `x ≡ r_i (mod m_i)` має єдиний розв'язок за модулем `M = Π m_i`: `crt([2, 3, 2], [3, 5, 7])` = `{ x: 23, M: 105 }` (класична задача Сунь-цзи: `x ≡ 2 (3), 3 (5), 2 (7)` → `23`).

## 6. Бінарний НСД (алгоритм Штейна)

Замінює ділення на зсуви й віднімання (швидше на апаратурі без швидкого ділення): якщо обидва парні — виносимо множник 2; якщо одне парне — ділимо його на 2; якщо обидва непарні — `gcd(a, b) = gcd(|a − b|, min(a, b))`. `binaryGcd(48, 18)` = `6`. Складність `O(log² n)` бітових операцій (`O(n²)` для `n`-бітових).

## 7. Застосування

- **Скорочення дробів:** `reduceFraction(48, 18)` = `[8, 3]`, `reduceFraction(4, -6)` = `[-2, 3]`.
- **Підрахунок взаємно простих із `n` (функція Ейлера `φ`):** `coprimeCountUpTo(12)` = `4`.
- **Пітагорові трійки (примітивні):** `pythagoreanTriples(30)` = `[[3, 4, 5], [5, 12, 13], [8, 15, 17], [7, 24, 25], [20, 21, 29]]` (`gcd(m, n) = 1`, різної парності).
- **Періодичні події:** збіг через `lcm` періодів (`lcmArray([4, 6, 10])` = `60`).
- **Розподіл на однакові групи:** найбільша кількість однакових груп = `gcd` кількостей; нормалізація співвідношень: `ratioOfRange([12, 18, 30])` = `[2, 3, 5]`.
- **Криптографія:** RSA, обернені елементи, алгоритм Діффі—Геллмана (за модулем).
- **Годинники/кільця зубчастих коліс, музичні інтервали, задачі про наливання води (кратність `gcd`).**

## 8. Типові помилки

1. **Від'ємні числа:** `%` у JS зберігає знак; беріть `Math.abs`.
2. **`gcd(0, 0)`:** результат `0` — окремий випадок при діленні.
3. **`lcm` без ділення спершу** — переповнення `a·b` (`2⁵³`).
4. **`modInverse` без перевірки `gcd = 1`** — хибна відповідь.
5. **Нормалізація результату `x % m`** у розширеному Евкліді: `((x % m) + m) % m`.
6. **Рекурсивний** варіант на великих числах — глибина невелика (~92), безпечно.
7. **Числа > 2⁵³** — `BigInt`.
8. **Діофантове рівняння:** забути перевірку `c % g`.
9. **Плутанина `Math.floor` і `Math.trunc`** для від'ємних у розширеному алгоритмі.

## 9. Алгоритм (стисло)

1. Поки `b ≠ 0`: `(a, b) = (b, a mod b)`.
2. Результат `a`. Розширена версія повертає також `x, y`, що `a·x + b·y = gcd`.
3. Час `O(log min(a, b))`.

   function solveLinearCongruence(a, b, m) {
     const g = gcd(a, m);
     if (b % g !== 0) return null;
     const a1 = a / g, b1 = b / g, m1 = m / g;

36. Приклад розв'язку:

   ```js
   function eulerPhi(n) {
     let r = n;
     for (let p = 2; p * p <= n; p++) if (n % p === 0) {
       while (n % p === 0) n /= p;
       r -= r / p;
     }
     if (n > 1) r -= r / n;
     return r;
   }
   ```

   Виклик `[eulerPhi(12), eulerPhi(13), eulerPhi(36), eulerPhi(1)]` → `[4, 12, 12, 1]`

37. Приклад розв'язку:

   ```js
   function farey(n) {
     const res = [];
     for (let b = 1; b <= n; b++) for (let a = 0; a <= b; a++) if (gcd(a, b) === 1) res.push([a, b]);
     res.sort((x, y) => x[0] * y[1] - y[0] * x[1]);
     return res.map(([a, b]) => a + '/' + b);
   }
   ```

40. Приклад розв'язку:

   ```js
   function gcdBySubtraction(a, b) {
     let steps = 0;
     while (a !== b) {
       if (a > b) a -= b;
       else b -= a;
       steps++;
     }
     return [a, steps];
   }
   ```

41. Будь-який спільний дільник `a` і `b` ділить `a − k·b`, зокрема `a mod b`; і навпаки, спільний дільник `b` та `a mod b` ділить `a = k·b + (a mod b)`. Отже, множини спільних дільників пар збігаються, як і найбільший з них.
42. `O(log min(a, b))` кроків. Найгірший випадок — послідовні числа Фібоначчі: кожен залишок є попереднім числом Фібоначчі, тому кроків стільки, скільки індекс у послідовності (теорема Ламе).
43. Для цілих `a, b` існують цілі `x, y`, що `a·x + b·y = gcd(a, b)`. Розширений алгоритм Евкліда знаходить їх, піднімаючись із рекурсії.
44. Якщо `gcd(a, m) = 1`, розширений Евклід дає `a·x + m·y = 1`, тому `a·x ≡ 1 (mod m)`, і `x mod m` — обернений елемент. Якщо `gcd ≠ 1`, оберненого не існує.
45. Тоді й лише тоді, коли `gcd(a, b)` ділить `c`. Один розв'язок знаходимо масштабуванням коефіцієнтів Безу на `c / gcd`, решта отримується додаванням `(b/g, −a/g)·k`.
46. `gcd(a, b) · lcm(a, b) = |a·b|`, тому `lcm(a, b) = a / gcd(a, b) · b`; ділення виконують першим, щоб уникнути переповнення добутку.
